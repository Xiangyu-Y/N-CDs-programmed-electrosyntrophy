# =========================
# Bubble heatmap: fixed dot size + significance stars
# =========================
library(ggplot2)
library(dplyr)
library(stringr)
library(scales)

data <- read.csv("D:/2025R/protein_expression_G_new.csv", check.names = FALSE)

data$Group <- factor(
  data$Group,
  levels = c("P+G-NCDs_vs_P+G", "P-NCDs_vs_P", "P+G_vs_P", "P+G-NCDs_vs_P-NCDs")
)

data$Accession <- factor(data$Accession, levels = rev(unique(data$Accession)))

data <- data %>%
  mutate(
    Pvalue = as.numeric(Pvalue),
    sig = case_when(
      is.na(Pvalue) ~ "",
      Pvalue == 0   ~ "",       
      Pvalue < 0.001 ~ "***",
      Pvalue < 0.01  ~ "**",
      Pvalue < 0.05  ~ "*",
      TRUE ~ ""
    )
  )


p <- ggplot(data, aes(x = Group, y = Accession)) +
  geom_point(
    aes(fill = Log2FC),
    shape = 21, color = "black", stroke = 0.25,
    size = 10, alpha = 0.85
  ) +
  geom_text(
    aes(label = sig),
    color = "grey20",
    size = 4,       
    fontface = "bold",
    vjust = 0.75      
  ) +
  scale_fill_gradient2(
    low = "#00B9F0",
    mid = "white",
    high = "#EB2879",
    midpoint = 0,
    limits = c(-6, 6),
    oob = squish,
    breaks = c(-6, -4, -2, 0, 2, 4, 6),
    labels = c("-6", "-4", "-2", "0", "2", "4", "6")
  ) +
  theme_minimal(base_size = 10) +
  theme(
    axis.text.x = element_text(angle = 90, hjust = 1, size = 14, face = "bold"),
    axis.text.y = element_text(size = 14, face = "bold"),
    panel.grid.major = element_line(color = "gray80", linewidth = 0.15),
    panel.grid.minor = element_blank(),
    legend.title = element_text(size = 9),
    legend.text = element_text(size = 8),
    plot.title = element_text(size = 10, face = "bold", hjust = 0.5, margin = margin(b = 10)),
    axis.ticks = element_blank(),
    panel.background = element_rect(fill = "white", color = NA),
    plot.background = element_rect(fill = "white", color = NA),
    panel.border = element_blank()
  ) +
  labs(
    x = NULL,
    y = NULL,
    fill = "Log2(FC)"
  )

p
